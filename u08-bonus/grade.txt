			players.add(game.createWarrior(100, 100 + d));
		}

		assertEquals(100, cleric.getHealth());
		assertEquals(100, cleric.getPosition());
		for (int i = 0; i < players.size(); ++i) {
			int d = i - 7;
			assertEquals(100, players.get(i).getHealth());
			assertEquals(100 + d, players.get(i).getPosition());
		}

		assertTrue(cleric.scheduleAction(Action.SUMMON));

		game.advanceTurn();
		game.advanceTurn();
		game.advanceTurn();

		assertEquals(100, cleric.getHealth());
		assertEquals(100, cleric.getPosition());
		for (int i = 0; i < players.size(); ++i) {
			int d = i - 7;
			assertEquals(100, players.get(i).getHealth());
			if (Math.abs(d) < 3 || Math.abs(d) > 5) {
				assertEquals(100 + d, players.get(i).getPosition());

			} else {
				assertEquals(100, players.get(i).getPosition());
			}
		}

	}

	@Test
	public void testEdgeCase04() {

		// Summoning on edge of board

		Game game = new Game();

		Human jester1 = game.createJester(100, 9);
		Human jester2 = game.createJester(100, 991);
		Human warrior1 = game.createWarrior(100, 10);
		Human warrior2 = game.createWarrior(100, 990);
		Human cleric1 = game.createCleric(100, 6);
		Human cleric2 = game.createCleric(100, 994);

		assertTrue(warrior1.scheduleAction(Action.ATTACK));
		assertTrue(warrior2.scheduleAction(Action.ATTACK));
		assertTrue(cleric1.scheduleAction(Action.SUMMON));
		assertTrue(cleric2.scheduleAction(Action.SUMMON));

		game.advanceTurn();

		assertEquals(90, jester1.getHealth());
		assertEquals(90, jester2.getHealth());

		game.advanceTurn();

		assertEquals(9, jester1.getPosition());
		assertEquals(991, jester2.getPosition());
		assertEquals(10, warrior1.getPosition());
		assertEquals(990, warrior2.getPosition());
		assertEquals(6, cleric1.getPosition());
		assertEquals(994, cleric2.getPosition());

		game.advanceTurn();

		assertEquals(6, jester1.getPosition());
		assertEquals(994, jester2.getPosition());
		assertEquals(6, warrior1.getPosition());
		assertEquals(994, warrior2.getPosition());
		assertEquals(6, cleric1.getPosition());
		assertEquals(994, cleric2.getPosition());

	}
