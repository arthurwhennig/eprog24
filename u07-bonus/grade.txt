                {1, 1, 2, 1, 1},
                {-1, -2, -1, -2, -1},
        };
        assertArrayEquals(intermediateBoard, startBoard);

        Minesweeper.gameTurn(intermediateBoard, 4, 2);

        int[][] expected = {
                {-1, -2, -2, 1, 0},
                {1, 2, 2, 1, 0},
                {0, 0, 0, 0, 0},
                {1, 1, 2, 1, 1},
                {-1, -2, 2, -2, -1},
        };
        assertArrayEquals(expected, intermediateBoard);
    }
